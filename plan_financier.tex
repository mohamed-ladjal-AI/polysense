\section{Synthèse — pour le dossier 1275}
% =====================================================================

\begin{itemize}\itemsep2pt
  \item \textbf{Objet social :} conception, développement et commercialisation de logiciels d'aide à la supervision industrielle (capteurs logiciels / soft-sensors), avec une première application dans l'extrusion du PVC en Algérie.
  \item \textbf{Ancrage technique :} mémoire de Master (université algérienne, 2025-2026) ayant produit un pipeline ML validé (F1=0{,}85 sur jeu de test, validé sur jeu industriel tiers).
  \item \textbf{Modèle de revenu :} licence annuelle par ligne d'extrusion, 200\,k~DZD (1\,500~USD) — inférieur au coût d'un quart de travail d'un ingénieur qualité.
  \item \textbf{Traction visée :} 1 pilote N, 3 clients payants N+1, 40 lignes servies N+3, 95 lignes N+4.
  \item \textbf{Emplois créés :} 2 en N, 6 en N+4.
  \item \textbf{Investissement initial :} 1{,}1~M~DZD ($\sim$8\,k~USD), intégralement couvert par le fondateur et le love money.
  \item \textbf{Année de rentabilité :} N+3 (seuil opérationnel atteint, résultat positif en N+4).
  \item \textbf{Demande 1275 :} label startup + accompagnement (mentorat, accès au marché public, exonérations fiscales 4 ans).
\end{itemize}
